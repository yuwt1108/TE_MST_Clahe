cmake_minimum_required(VERSION 3.16)
project(te_mst_demo LANGUAGES CXX)

find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui)
add_executable(demo_photo demo_photo.cpp te_mst.cpp te_mst.h)
add_executable(demo_video demo_video.cpp te_mst.cpp te_mst.h)
foreach(target IN ITEMS demo_photo demo_video)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    target_include_directories(${target} PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(${target} PRIVATE ${OpenCV_LIBS})
endforeach()
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT demo_video)
set_property(TARGET demo_photo demo_video PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
if(TARGET opencv_world)
    set_property(TARGET demo_photo demo_video PROPERTY VS_DEBUGGER_ENVIRONMENT
        "PATH=$<TARGET_FILE_DIR:opencv_world>;%PATH%")
endif()
